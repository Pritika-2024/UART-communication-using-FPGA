`timescale 1ns / 1ps
module rx
 #(parameter clks_per_bit=5208)
( input i_clock,
  input i_rx_serial,
  output o_rx_dv,
  output[7:0] o_rx_byte
 );
	 
parameter s_idle=3'b000;
parameter s_start=3'b001;
parameter s_data=3'b010;
parameter s_stop=3'b011;
parameter s_cleanup=3'b100;

reg r_data_byte_r = 1'b1;
reg r_data        = 1'b1;

reg[12:0] r_clk_count=0;
reg[2:0] r_bit_index=0;
reg[7:0] r_dataByte=0;
reg r_dv=0; 
reg[2:0] r_state=s_idle;

always @(posedge i_clock)
  begin
   r_data_byte_r <= i_rx_serial;
   r_data   <= r_data_byte_r;
  end

always @(posedge i_clock)
  begin 
  case (r_state)
   s_idle:
	 begin
	 r_dv<=1'b0;
	 r_clk_count<=13'd0;
	 r_bit_index<=3'd0;
	 if (r_data ==1'b1 && r_data_byte_r==1'b0)
	   begin
	   r_clk_count<=13'd0;
	   r_state<=s_start;
		end
	 else 
	   r_state<=s_idle;
	 end
	 
	 s_start:
	  begin
	  if (r_clk_count == (clks_per_bit >> 1))
	   begin
		if (r_data==1'b0)
		 begin
		 r_clk_count<=13'd0;
		 r_state<=s_data;
		 end
		else
		r_state<=s_idle;
	  end
	  else 
	   begin
	   r_clk_count<=r_clk_count+13'd1;
	   r_state<=s_start;
	   end
	  end
	  
	  s_data:
	  begin
	  if(r_clk_count < (clks_per_bit-1))
	   begin
		r_clk_count<=r_clk_count+13'd1;
		r_state<=s_data;
		end
	  else
	   begin
		r_clk_count <= 13'd0;
		r_dataByte[r_bit_index] <= r_data;
	
		if (r_bit_index < 3'd7)
		 begin
		 r_bit_index<=r_bit_index+3'd1;
		 r_state<=s_data;
		 end
		else
		 begin 
		 r_bit_index <=3'd0;
		 r_state<=s_stop;
		 end
		end
     end	
	  
     s_stop:
      begin
		if(r_clk_count < (clks_per_bit-1))
	    begin
		  r_clk_count<=r_clk_count+13'd1;
		  r_state<=s_stop;
		 end
      else
		 begin
		  r_dv<=1'b1;
		  r_clk_count<=13'd0;
		  r_state<=s_cleanup;
      end	
     end	
	  
    s_cleanup:
     begin
		r_state<=s_idle;
		r_dv<=1'b0;
     end		
	 default:
	 r_state<=s_idle;
	 endcase
	end
assign o_rx_byte = r_dataByte;
assign o_rx_dv = r_dv;
endmodule
