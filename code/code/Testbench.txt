`timescale 1ns / 1ps

module uart_tb;
  parameter clk_period=10;//10ns->10^-8s->10^8hz=100Mhz
	// Inputs
	reg i_clock;
	reg i_rx_serial;
	// Outputs
	wire o_tx_serial;
	
	initial i_clock = 0;
   always #(clk_period/2)
	    i_clock = ~i_clock;
		 
	// Instantiate the Unit Under Test (UUT)
	top uut (
		.i_clock(i_clock), 
		.i_rx_serial(i_rx_serial), 
		.o_tx_serial(o_tx_serial)
	);
	
  task send_byte;
  input[7:0] data;
  integer i;
  begin
  //start bit
   i_rx_serial = 1'b0;
	#(5208 * clk_period);
	//data bit
	for(i=0;i<8;i=i+1)
	  begin i_rx_serial =data[i];
	  #(5208 * clk_period);
	  end
	//stop bit
	i_rx_serial=1'b1;
	#(5208 * clk_period);
	end
endtask
	 
	initial begin
	   i_rx_serial=1'b1;
	   #1000;
		// Initialize Inputs
      send_byte(8'h3F); //0(start) 1(lsb)1111100(msb) 1(stop)
      #100;
	end
      
endmodule
