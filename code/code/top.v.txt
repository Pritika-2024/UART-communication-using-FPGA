`timescale 1ns / 1ps

module top
(
    input i_clock,        
    input i_rx_serial,    
    output o_tx_serial,   
    output [7:0] o_seg,   
    output [2:0] o_sel   
);
    wire w_dv;
    wire [7:0] w_raw_byte;
    wire w_tx_done;
    reg r_tx_dv = 0;
    reg [7:0] r_captured_byte = 8'h20; 
    wire w_parity;
	 
    rx #(.clks_per_bit(5208)) UART_RX (
        .i_clock(i_clock),
        .i_rx_serial(i_rx_serial),
        .o_rx_dv(w_dv),
        .o_rx_byte(w_raw_byte)
    );

    assign w_parity = ^w_raw_byte; 

    always @(posedge i_clock) begin
        if (w_dv == 1'b1) begin
            r_tx_dv <= 1'b1;
            r_captured_byte <= w_raw_byte;
        end else if (w_tx_done == 1'b1) begin
            r_tx_dv <= 1'b0;
        end
    end
 
    tx #(.clks_per_bit(5208)) UART_TX (
        .i_clock(i_clock),
        .i_tx_dv(r_tx_dv),
        .i_tx_byte(r_captured_byte),
        .o_tx_active(),
        .o_tx_serial(o_tx_serial),
        .o_tx_done(w_tx_done)
    );

    reg [16:0] r_refresh_cnt = 0;
    reg [7:0] r_active_char;
    always @(posedge i_clock) r_refresh_cnt <= r_refresh_cnt + 1;
    assign o_sel = (r_refresh_cnt[16]) ? 3'b110 : 3'b101; 

    always @(*) begin
        if (r_refresh_cnt[16]) 
            r_active_char = r_captured_byte;        
        else 
            r_active_char = w_parity ? 8'h31 : 8'h30;
    end

    // Mapping format: 8'b{h, g, f, e, d, c, b, a}
    reg [7:0] r_seg_out;
    always @(*) begin
        case(r_active_char)
            
            8'h30: r_seg_out = 8'b11000000; // '0'
            8'h31: r_seg_out = 8'b11111001; // '1'
            8'h32: r_seg_out = 8'b10100100; // '2'
            8'h33: r_seg_out = 8'b10110000; // '3'
            8'h34: r_seg_out = 8'b10011001; // '4'
            8'h35: r_seg_out = 8'b10010010; // '5'
            8'h36: r_seg_out = 8'b10000010; // '6'
            8'h37: r_seg_out = 8'b11111000; // '7'
            8'h38: r_seg_out = 8'b10000000; // '8'
            8'h39: r_seg_out = 8'b10010000; // '9'
            
         
            8'h41, 8'h61: r_seg_out = 8'b10001000; // A / a
            8'h42, 8'h62: r_seg_out = 8'b10000011; // b
            8'h43, 8'h63: r_seg_out = 8'b11000110; // C / c
            8'h44, 8'h64: r_seg_out = 8'b10100001; // d
            8'h45, 8'h65: r_seg_out = 8'b10000110; // E / e
            8'h46, 8'h66: r_seg_out = 8'b10001110; // F / f
            8'h48, 8'h68: r_seg_out = 8'b10001011; // h
            8'h4C, 8'h6C: r_seg_out = 8'b11000111; // L / l
            8'h4F, 8'h6F: r_seg_out = 8'b10100011; // o
            8'h50, 8'h70: r_seg_out = 8'b10001100; // P / p
            8'h53, 8'h73: r_seg_out = 8'b10010010; // S / s 
            8'h55, 8'h75: r_seg_out = 8'b11000001; // U / u
            8'h59, 8'h79: r_seg_out = 8'b10010001; // Y / y
            8'h20:        r_seg_out = 8'b10111111; 

            default:      r_seg_out = w_parity ? 8'b11111001 : 8'b11000000; 
        endcase
    end

    assign o_seg = r_seg_out;

endmodule
