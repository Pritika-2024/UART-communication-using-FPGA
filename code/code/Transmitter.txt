`timescale 1ns / 1ps
module tx
 #(parameter clks_per_bit = 5208)
(
  input        i_clock,  //from ucf
  input        i_tx_dv, ///1 bit
  input  [7:0] i_tx_byte,//8 bits
  output       o_tx_active,
  output reg   o_tx_serial,
  output       o_tx_done
);

parameter s_idle    = 3'b000;
parameter s_start   = 3'b001;
parameter s_data    = 3'b010;
parameter s_stop    = 3'b011;
parameter s_cleanup = 3'b100;

reg [2:0]  r_state      = s_idle;
reg [13:0] r_clk_count  = 0;    
reg [2:0]  r_bit_index  = 0;
reg [7:0]  r_tx_data    = 0;
reg        r_tx_done    = 0;
reg        r_tx_active  = 0;

always @(posedge i_clock)
begin
  case (r_state)

    s_idle:
    begin
      o_tx_serial <= 1'b1;   
      r_tx_done   <= 1'b0;
      r_clk_count <= 14'd0;
      r_bit_index <= 3'd0;

      if (i_tx_dv == 1'b1)
      begin
        r_tx_active <= 1'b1;
        r_tx_data   <= i_tx_byte;
        r_state     <= s_start;
      end
    end


    s_start:
    begin
      o_tx_serial <= 1'b0;

      if (r_clk_count < clks_per_bit - 1)
        r_clk_count <= r_clk_count + 14'd1;
      else
      begin
        r_clk_count <= 14'd0;
        r_state     <= s_data;
      end
    end

    s_data:
    begin
      o_tx_serial <= r_tx_data[r_bit_index];

      if (r_clk_count < clks_per_bit - 1)
        r_clk_count <= r_clk_count + 14'd1;
      else
      begin
        r_clk_count <= 14'd0;

        if (r_bit_index < 3'd7)
		  begin
          r_bit_index <= r_bit_index + 3'd1;
			 r_state  <=  s_data;
		  end
        else
        begin
          r_bit_index <= 3'd0;
          r_state     <= s_stop;
        end
      end
    end

    s_stop:
    begin
      o_tx_serial <= 1'b1;

      if (r_clk_count < clks_per_bit - 1)
        r_clk_count <= r_clk_count + 14'd1;
      else
      begin
        r_tx_done   <= 1'b1;
        r_clk_count <= 14'd0;
        r_state     <= s_cleanup;
        r_tx_active <= 1'b0;
      end
    end

    s_cleanup:
    begin
      r_tx_done <= 1'b1;
      r_state   <= s_idle;
    end

    default:
      r_state <= s_idle;

  endcase
end

assign o_tx_active = r_tx_active;
assign o_tx_done   = r_tx_done;

endmodule
